    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/9.23.0/firebase-app.js";
        import { getAuth, onAuthStateChanged, signOut } from "https://www.gstatic.com/firebasejs/9.23.0/firebase-auth.js";
        import { getFirestore, doc, getDoc } from "https://www.gstatic.com/firebasejs/9.23.0/firebase-firestore.js";

        // Configuration Firebase officielle de DAKPRO ÉLITE
        const firebaseConfig = {
          apiKey: "AIzaSyDR7INHKaazaqZt-xIcjk10JFiy58uXKO8",
          authDomain: "dakproelite.firebaseapp.com",
          databaseURL: "https://dakproelite-default-rtdb.firebaseio.com",
          projectId: "dakproelite",
          storageBucket: "dakproelite.firebasestorage.app",
          messagingSenderId: "580591769206",
          appId: "1:580591769206:web:4f67f8aadbf3d051087157"
        };

        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);

        // Fonction globale de déconnexion propre
        window.deconnecter = async () => {
            try {
                localStorage.removeItem('user_role');
                localStorage.removeItem('user_profile_url');
                await signOut(auth);
                window.location.replace("connexion.html");
            } catch (error) {
                console.error("Erreur lors de la déconnexion :", error);
            }
        };

        // Écoute de l'état d'authentification en temps réel
        onAuthStateChanged(auth, async (user) => {
            if (!user) {
                // Si l'utilisateur n'est pas connecté, renvoi direct vers la page de connexion
                window.location.replace("connexion.html");
                return;
            }

            try {
                // Interrogation de la collection principale "utilisateurs"
                let userDoc = await getDoc(doc(db, "utilisateurs", user.uid));
                
                // Si non trouvé dans "utilisateurs", tentative de secours dans "vendeurs"
                if (!userDoc.exists()) {
                    userDoc = await getDoc(doc(db, "vendeurs", user.uid));
                }
                
                if (userDoc.exists()) {
                    const userData = userDoc.data();
                    const role = userData.role ? userData.role.toLowerCase().trim() : 'acheteur';
                    redirigerSelonRole(role);
                } else {
                    // Si aucun document n'est trouvé, redirection par défaut vers l'espace acheteur
                    window.location.replace("acheteur.html");
                }
            } catch (error) {
                console.error("Erreur de récupération du rôle depuis Firestore :", error);
                window.location.replace("acheteur.html");
            }
        });

        // Fonction d'aiguillage centralisée vers les fichiers exacts du projet
        function redirigerSelonRole(role) {
            let pageDestination = "acheteur.html";

            switch (role) {
                case 'vendeur':
                case 'e-commerçant':
                case 'e-commercant':
                    pageDestination = "vendeur.html"; // Pointe vers ton fichier unique vendeur.html
                    break;
                case 'livreur':
                    pageDestination = "livreur.html";
                    break;
                case 'admin':
                case 'administrateur':
                    pageDestination = "administration.html";
                    break;
                case 'acheteur':
                default:
                    pageDestination = "acheteur.html";
                    break;
            }

            // Sauvegarde dans le stockage local
            localStorage.setItem('user_role', role);
            localStorage.setItem('user_profile_url', pageDestination);

            // Redirection vers la page correspondante
            window.location.replace(pageDestination);
        }
    </script>
